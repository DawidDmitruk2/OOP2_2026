Tworzenie i metody odczytu
import java.util.Scanner;

// Scanner z konsoli (System.in)
Scanner sc = new Scanner(System.in);

// Metody odczytu jednej wartości:
int     n   = sc.nextInt();       // czyta liczbę całkowitą
double  d   = sc.nextDouble();    // czyta liczbę zmiennoprzecinkową
long    l   = sc.nextLong();      // czyta long
boolean b   = sc.nextBoolean();   // czyta "true" / "false"
String  tok = sc.next();          // czyta jedno słowo (do białego znaku)
String  lin = sc.nextLine();      // czyta CAŁĄ linię (z spacjami)

// Sprawdzanie czy jest coś do odczytu:
sc.hasNext();           // czy jest kolejny token
sc.hasNextLine();       // czy jest kolejna linia
sc.hasNextInt();        // czy następny token to int

// Zawsze zamykaj Scanner po użyciu
sc.close();

Pełny wzorzec fromCsv() przez Scanner
public static Produkt fromCsv(Path sciezka) {
    try (Scanner sc = new Scanner(sciezka)) {

        // linia 0: nazwa produktu
        String nazwa = sc.nextLine().trim();

        // linia 1: nagłówek — pomijamy
        sc.nextLine();

        // linie 2+: dane
        List<double[]> wiersze = new ArrayList<>();
        while (sc.hasNextLine()) {
            String linia = sc.nextLine();
            if (linia.isBlank()) continue;

            String[] col = linia.split(";");
            double[] ceny = new double[col.length];
            for (int i = 0; i < col.length; i++)
                ceny[i] = Double.parseDouble(col[i].trim());
            wiersze.add(ceny);
        }

        return new Produkt(nazwa, wiersze);

    } catch (IOException e) {
        throw new RuntimeException("Błąd pliku: " + sciezka, e);
    }
}

}
