import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

public class CovidDataParser {

    public static void main(String[] args) {
        // Zastąp poniższe ścieżki nazwami swoich plików
        String casesFile = "confirmed_cases.csv";
        String deathsFile = "deaths.csv";
        String outputFile = "out.txt";

        // Używamy bloku try-with-resources.
        // Gwarantuje to, że pliki zostaną automatycznie zamknięte po zakończeniu pracy.
        try (
            BufferedReader casesReader = new BufferedReader(new FileReader(casesFile));
            BufferedReader deathsReader = new BufferedReader(new FileReader(deathsFile));
            BufferedWriter writer = new BufferedWriter(new FileWriter(outputFile))
        ) {
            // 1. OMINIĘCIE NAGŁÓWKÓW
            // Pierwsze dwa wiersze to nazwy państw i prowincji, więc pomijamy je czytając "w próżnię".
            for (int i = 0; i < 2; i++) {
                casesReader.readLine();
                deathsReader.readLine();
            }

            // 2. PRZYGOTOWANIE FORMATERÓW DAT
            DateTimeFormatter inputFormatter = DateTimeFormatter.ofPattern("M/d/yy");
            DateTimeFormatter outputFormatter = DateTimeFormatter.ofPattern("d.MM.yy");

            String casesLine;
            String deathsLine;

            // 3. ODCZYT LINIA PO LINII
            // Ponieważ oba pliki mają tę samą liczbę wierszy, czytamy je równocześnie.
            while ((casesLine = casesReader.readLine()) != null &&
                   (deathsLine = deathsReader.readLine()) != null) {

                // Dzielimy wiersz na poszczególne kolumny przy użyciu średnika
                String[] casesData = casesLine.split(";");
                String[] deathsData = deathsLine.split(";");

                // Pobieramy datę z pierwszej kolumny i zmieniamy jej format
                String rawDate = casesData[0];
                LocalDate date = LocalDate.parse(rawDate, inputFormatter);
                String formattedDate = date.format(outputFormatter);

                // Zakładamy, że pobieramy statystyki dla jakiegoś państwa.
                // W uproszczeniu pobieramy pierwszą kolumnę danych po dacie (indeks 1 - Afganistan).
                int casesCount = Integer.parseInt(casesData[1]);
                int deathsCount = Integer.parseInt(deathsData[1]);

                // 4. ZAPIS DO PLIKU WYNIKOWEGO
                // Zapisujemy: data \t przypadki \t zgony
                writer.write(formattedDate + "\t" + casesCount + "\t" + deathsCount);
                writer.newLine(); // Konieczne, by przejść do nowego wiersza
            }

            System.out.println("Zakończono pobieranie i wysyłanie danych do pliku!");

        } catch (IOException e) {
            System.err.println("Wystąpił błąd podczas pracy z plikami: " + e.getMessage());
        }
    }
}
